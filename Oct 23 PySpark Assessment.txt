df_categorized = df \
    .withColumn("efficiency", (col("units_produced") / col("units_planned")) * 100) \
    .withColumn("efficiency_category",
        when(col("efficiency") >= 100, "Above Target")
        .when(col("efficiency") >= 95, "On Target")
        .when(col("efficiency") >= 85, "Below Target")
        .otherwise("Critical")
    ) \
    .withColumn("defect_category",
        when(col("defective_units") == 0, "Perfect")
        .when((col("defective_units") >= 1) & (col("defective_units") <= 10), "Normal")
        .when((col("defective_units") >= 11) & (col("defective_units") <= 25), "Attention")
        .otherwise("High Defect")
    ) \
    .withColumn("machine_condition",
        when((col("temperature_c") > 85) | (col("vibration_mm_s") > 5.0), "High")
        .otherwise("Normal")
    )


df_categorized.groupBy("efficiency_category").count().show()
df_categorized.groupBy("defect_category").count().show()
df_categorized.groupBy("machine_condition").count().show()
