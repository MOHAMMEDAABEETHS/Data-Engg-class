cleaned_shipments_df = (
    shipments_df
    .na.fill(0, subset=["shipping_cost_inr", "package_weight_kg"])
)


processed_pipeline_df = (
    cleaned_shipments_df
    .withColumn("booking_dt", F.to_date("booking_date"))
    .withColumn("actual_dt", F.to_date("actual_delivery_date"))
    .withColumn("promised_dt", F.to_date("promised_delivery_date"))
    .withColumn(
        "delivery_duration_days",
        F.datediff(F.col("actual_dt"), F.col("booking_dt"))
    )
    .withColumn(
        "is_late",
        F.when(F.col("actual_dt") > F.col("promised_dt"), 1).otherwise(0)
    )
    .withColumn(
        "is_on_time",
        F.when((F.col("actual_dt").isNotNull()) & (F.col("actual_dt") <= F.col("promised_dt")), 1).otherwise(0)
    )
    .withColumn(
        "shipment_profit",
        F.col("shipping_fee_inr") - F.col("shipping_cost_inr")
    )
    .join(warehouse_df, "warehouse_id")
    .join(carrier_df, "carrier_id")
)

aggregated_pipeline_df = (
    processed_pipeline_df
    .groupBy("region", "carrier_id", "carrier_name")
    .agg(
        F.sum("shipping_fee_inr").alias("total_revenue"),
        F.sum("shipment_profit").alias("total_profit"),
        F.count("shipment_id").alias("total_shipments"),
        (F.sum("is_late") / F.count("shipment_id") * 100).alias("late_delivery_percentage"),
        (F.sum("is_on_time") / F.count("shipment_id") * 100).alias("on_time_delivery_percentage")
    )
)

window_ranking = Window.partitionBy("region").orderBy(F.desc("on_time_delivery_percentage"))
ranked_pipeline_df = aggregated_pipeline_df.withColumn(
    "carrier_rank_by_ontime",
    F.dense_rank().over(window_ranking)
)


ranked_pipeline_df.cache()
print("Final Report Record Count:", ranked_pipeline_df.count())
